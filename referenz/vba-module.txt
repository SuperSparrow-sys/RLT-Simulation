Sub Speicher()
'Überträgt den Endwert auf den Startwert
    With Application
        .Iteration = False
    End With
    i = 0
    For Each c In Sheets("Anlage").Range("Speicherwert")
        If i = 1 Then
           c.Value = endwert
           i = 0
        Else
          endwert = c.Value
          i = 1
        End If
    Next c
    With Application
        .Iteration = True
    End With


End Sub
